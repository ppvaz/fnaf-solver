Q = '"kind","ts_ns","dur_ns","name","thread_name","process_name","track_name"\n'

def row(kind, ts, name, dur=1):
    q = lambda v: '"' + str(v).replace('"', '""') + '"'
    return ','.join(q(v) for v in (kind, ts, dur, name, 'main', 'com.scottgames.fnaf2', 'game')) + '\n'
P = lambda a: f'publishMotionEvent(inputChannel=game, action={a})'
R = lambda s: f'receiveMessage(inputChannel=game, seq={s}, type=MOTION)'
F = lambda s: f'receiveMessage(inputChannel=game, seq={s}, type=FINISHED)'
D = lambda a, dev: f'dispatchInputEvent MotionEvent ACTION_{a} deviceId={dev} source=0x1002 historySize=0'
def synthetic(actions, injected=True, dev=9, finish=True):
    rows = []
    if injected:
        for ts, a, s in [(1, 'DOWN', '0x1'), (4, 'MOVE', '0x2'), (7, 'UP', '0x3')]:
            rows += [row('publish', ts, P(a)), row('receive', ts + 1, R(s)), row('finish', ts + 2, F(s)), row('dispatch', ts, D(a, -1))]
    for k, (ts, a) in enumerate(actions):
        s = hex(0x10 + k)
        rows += [row('publish', ts, P(a)), row('receive', ts + 1, R(s)), row('dispatch', ts, D(a, dev))]
        if finish: rows.append(row('finish', ts + 2, F(s)))
    return Q + ''.join(rows)
four = [(100_000_000, 'DOWN'), (110_000_000, 'UP'), (300_000_000, 'DOWN'), (310_000_000, 'UP'), (500_000_000, 'DOWN'), (510_000_000, 'UP'), (700_000_000, 'DOWN'), (710_000_000, 'UP')]

  'noaction': Q + row('publish', 1, 'publishMotionEvent(inputChannel=game, act=DOWN)') + row('receive', 2, R('0x1')),

}
clocks = {

def tsv(rows, header='# schema=fnaf2-frame-trace-v3 fps=60 note=a=b  bare __proto__=x =y', cols='seq\timage_ns\tinterval_ns\tmask_downstroke\tmonitor_downstroke\textra', blank=True):
    body = '\n'.join(f'{k + 1}\t{t * 1_000_000}\t0\t{m}\t{n}\tz' for k, (t, m, n) in enumerate(rows))
    return header + '\n' + cols + '\n' + ('\n \t\n' if blank else '') + body + '\n'
states = [(0, 142, 144), (90, 142, 144), (120, 4, 24), (140, 0, 144), (320, 0, 0), (360, 142, 144), (520, 4, 20), (550, 142, 20), (720, 4, 0), (740, 142, 0), (800, 142, 144)]
frames = {
  'f.tsv': tsv(states), 'f-nofinal.tsv': tsv(states[:-1]), 'f-unsorted.tsv': tsv(list(reversed(states)) + [(800, 142, 144), (801, -1, 144)]),
  'f-bad-schema.tsv': tsv(states, header='# schema=other'), 'f-missing.tsv': tsv(states, cols='seq\timage_ns'), 'f-short.tsv': '# schema=fnaf2-frame-trace-v3\nseq\n',
  'f-count.tsv': tsv(states) + '1\t2\n', 'f-num.tsv': tsv(states) + '1\tx\t0\t0\t0\tz\n', 'f-blank.tsv': tsv([], blank=True),
  'f-crlf.tsv': tsv(states).replace('\n', '\r\n'), 'f-uni.tsv': tsv(states) + '\u0669\u0669\t\u0669\u0660\u0660\u0660\u0660\u0660\u0660\u0660\u0660\t0\t 1 \t1_0\tz\n', 'f-one.tsv': tsv([(5, 142, 144)]),
  'f-25.tsv': tsv([(0, 142, 144), (30, 142, 144), (40, 4, 144), (100, 4, 144)]),
}
binary = {'f-bom.tsv': b'\xef\xbb\xbf' + tsv(states).encode(), 'f-utf8.tsv': tsv(states).encode() + b'1\t\xff\n', 'f-surrogate.tsv': tsv(states).encode() + b'\xed\xa0\x80\n'}

cases += [('syn', 'clock0', 'f.tsv', ['--max-wait-ms', '50']), ('syn', 'clock0', 'f.tsv', ['--max-wait-ms', '0']), ('syn', 'clock0', 'f.tsv', ['--max-wait-ms', '-1']),
  ('syn', 'clock0', 'f.tsv', ['--max-wait-ms', 'nan']), ('syn', 'clock0', 'f.tsv', ['--max-wait-ms', 'inf']), ('none', 'clock0', 'f.tsv', ['--max-wait-ms', 'inf']),

  ('ENVPROC', 'clock0', 'f.tsv', []), ('PATHPROC', 'clock0', 'f.tsv', []), ('syn', 'clock0', 'f.tsv', ['--trace-processor', 'nope']), ('syn', 'clock0', './/f.tsv', ['--json'])]
same = 0

    for n, t in csvs.items(): open(os.path.join(d, n + '.csv'), 'w', newline='').write(t)

    proc = os.path.join(d, 'trace_processor')
    open(proc, 'w').write('#!/bin/sh\n[ -n "$FAKE_SIG" ] && kill -9 $$\nif [ -n "$FAKE_RC" ]; then case "$3" in *clock_snapshot*) [ -n "$FAKE_CLOCKONLY" ] && { printf "$FAKE_ERR" >&2; exit "$FAKE_RC"; };; *) [ -z "$FAKE_CLOCKONLY" ] && { printf "$FAKE_ERR" >&2; exit "$FAKE_RC"; };; esac; fi\n'

            print('DIFF', csvname, clockname, frame, args, 'rc', out[0][1], out[1][1], '\n  py', a[k:k+2], out[0][2], '\n  ts', b[k:k+2], out[1][2])
